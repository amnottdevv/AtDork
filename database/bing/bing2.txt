# ============================================
# BING DORKS - ADVANCED OPERATORS
# ============================================
#
# BING OPERATOR EKSKLUSIF:
# 1. loc: / location: - Mencari berdasarkan lokasi geografis
# 2. ip: - Mencari website di alamat IP tertentu
# 3. domain: - Mencari domain tertentu (lebih akurat dari site:)
# 4. linkfromdomain: - Mencari halaman yang TIDAK ada link dari domain tertentu
# 5. language: - Mencari dalam bahasa tertentu
# 6. feed: - Mencari RSS/Atom feed
# 7. hasrelationship: - Mencari relasi antar website
# 8. prefer: - Meningkatkan bobot keyword tertentu
# 9. site: - (sama seperti Google tapi dengan tambahan)
#
# ============================================

# ============================================
# 1. loc: / location: - MENCARI BERDASARKAN LOKASI
# ============================================
#
# FUNGSI: Mencari website yang terasosiasi dengan lokasi geografis
# KEGUNAAN: Untuk pentest regional/target spesifik
# ============================================

# Cari website di Indonesia
location:Indonesia inurl:product.php?id=
location:Indonesia inurl:view.php?id=
location:Indonesia inurl:?page=
location:Indonesia inurl:?q=

# Cari website di Jakarta
location:Jakarta inurl:admin.php
location:Jakarta inurl:upload.php
location:Jakarta inurl:?id=

# Cari website di kota-kota besar Indonesia
location:Jakarta inurl:?page=
location:Surabaya inurl:?id=
location:Bandung inurl:?q=
location:Medan inurl:?cat=
location:Yogyakarta inurl:upload.php

# Cari website pemerintah berdasarkan lokasi
location:Indonesia site:.go.id inurl:?id=
location:Jakarta site:.go.id inurl:?page=

# Cari website pendidikan berdasarkan lokasi
location:Indonesia site:.ac.id inurl:?q=
location:Yogyakarta site:.ac.id inurl:?search=

# Cari website perusahaan berdasarkan lokasi
location:Indonesia site:.co.id inurl:?id=
location:Jakarta site:.co.id inurl:?cat=

# Cari website di negara ASEAN
location:Singapore inurl:?id=
location:Malaysia inurl:?page=
location:Thailand inurl:?q=

# Cari website global berdasarkan lokasi
location:United States inurl:product.php?id=
location:United Kingdom inurl:view.php?id=
location:Australia inurl:?id=

# ============================================
# 2. ip: - MENCARI BERDASARKAN ALAMAT IP
# ============================================
#
# FUNGSI: Mencari semua website yang berada di satu alamat IP (shared hosting)
# KEGUNAAN: Menemukan website lain di server yang sama untuk pentest
# ============================================

# Cari website di IP tertentu (ganti dengan IP target)
ip:8.8.8.8
ip:1.1.1.1
ip:192.168.1.1

# Cari website di IP server Indonesia
ip:103.8.12.100
ip:114.5.1.1
ip:180.240.0.1

# Cari website di IP + parameter rentan
ip:8.8.8.8 inurl:?id=
ip:8.8.8.8 inurl:?page=
ip:8.8.8.8 inurl:?q=

# Cari website di IP + file sensitif
ip:8.8.8.8 inurl:admin.php
ip:8.8.8.8 inurl:phpinfo.php
ip:8.8.8.8 contains:config

# Cari website di IP + CMS tertentu
ip:8.8.8.8 inurl:wp-content
ip:8.8.8.8 inurl:components

# Cari website di IP + error
ip:8.8.8.8 intext:"mysql"
ip:8.8.8.8 intext:"error"

# Cari semua website di shared hosting (IP server)
ip:103.8.12.100 inurl:?id=
ip:103.8.12.100 inurl:?q=
ip:103.8.12.100 inurl:upload.php

# ============================================
# 8. prefer: - MENINGKATKAN BOBOT KEYWORD
# ============================================
#
# FUNGSI: Memberi prioritas pada keyword tertentu
# KEGUNAAN: Meningkatkan akurasi hasil
# ============================================

# Prioritaskan parameter ID di hasil
prefer:inurl:?id= inurl:product.php
prefer:inurl:?page= inurl:view.php
prefer:inurl:?q= inurl:search.php

# Prioritaskan file sensitif
prefer:config inurl:config.php
prefer:backup inurl:backup.sql
prefer:admin inurl:admin.php

# Prioritaskan error SQL
prefer:mysql intext:"mysql" inurl:?id=
prefer:warning intext:"Warning" inurl:?page=
