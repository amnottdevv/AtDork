# KOMBINASI DENGAN OPERATOR LAIN

# Cari error SQL (indikasi celah)
inurl:product.php?id= intext:"mysql" 
inurl:product.php?id= intext:"sql" 
inurl:product.php?id= intext:"warning" 
inurl:product.php?id= intext:"error" 
inurl:product.php?id= intext:"failed" 
inurl:product.php?id= intext:"syntax" 

# Cari di domain tertentu (ganti target.com)
site:target.com inurl:product.php?id=
site:target.com inurl:view.php?id=
site:target.com inurl:?id=
site:target.com inurl:?cat=

# Cari berdasarkan ekstensi
inurl:product.php?id= ext:php
inurl:view.php?id= ext:php
inurl:?id= ext:php
inurl:?cat= ext:php

# Cari di judul tertentu
inurl:product.php?id= intitle:"product"
inurl:view.php?id= intitle:"view"
inurl:?id= intitle:"detail"
inurl:?cat= intitle:"category"

# Cari parameter numerik (rawan SQLi)
inurl:product.php?id=1
inurl:product.php?id=2
inurl:product.php?id=3
inurl:product.php?pid=1
inurl:product.php?pid=2
inurl:view.php?id=1
inurl:detail.php?id=1
inurl:item.php?id=1
inurl:news.php?id=1

# Cari parameter yang menggunakan angka di URL
inurl:"?id=" inurl:"=" intitle:"index"
inurl:"?cat=" inurl:"=" intitle:"index"
inurl:"?page=" inurl:"=" intitle:"index"

# Cari di website pemerintah Indonesia (contoh)
site:.go.id inurl:product.php?id=
site:.go.id inurl:view.php?id=
site:.go.id inurl:?id=
site:.go.id inurl:?cat=

# Cari di website pendidikan
site:.ac.id inurl:product.php?id=
site:.sch.id inurl:view.php?id=
site:.edu inurl:?id=

# Cari yang masih pakai parameter lama (versi rentan)
inurl:product.php?id= inurl:"mysql" inurl:"version"
inurl:view.php?id= intitle:"error" inurl:"sql"

# Cari parameter dengan tanda kutip (uji SQLi)
inurl:' inurl:product.php?id=
inurl:' inurl:view.php?id=
inurl:" inurl:?id=

# Cari parameter yang bisa di-order (order by)
inurl:product.php?id= order by
inurl:view.php?id= order by
inurl:?id= order by

# Cari parameter UNION (uji SQLi lanjutan)
inurl:product.php?id= union
inurl:view.php?id= union
inurl:?id= union
inurl:?cat= union

# Cari parameter dengan komentar SQL (-- , #)
inurl:product.php?id= --
inurl:view.php?id= --
inurl:?id= --
inurl:?cat= --
inurl:product.php?id= %23
inurl:view.php?id= %23

# Cari parameter AND/OR (uji boolean SQLi)
inurl:product.php?id= AND
inurl:view.php?id= AND
inurl:?id= AND
inurl:?cat= AND
inurl:product.php?id= OR
inurl:view.php?id= OR
